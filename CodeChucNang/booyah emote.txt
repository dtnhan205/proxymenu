[2] CODE MẪU ĐỘC LẬP (HELPER C#) - DỄ DÀNG COPY & PASTE VÀO MỌI SOURCE:
--------------------------------------------------------------------------------
using System;
using COW;
using COW.GamePlay;

public static class SuperEmoteHelper
{
    // 1. Kích hoạt Điệu nhảy Booyah / Super Emote
    public static void PlayBooyahEmote(Player localPlayer)
    {
        if (localPlayer == null) return;
        try
        {
            localPlayer.StartCheckBooyahEmote();
            localPlayer.PlayCarni25GPDanceEmote();
            localPlayer.RequestPlayBooyahSuperEmote(0);
        }
        catch (Exception)
        {
        }
    }

    // 2. Dừng Điệu nhảy (Trở về trạng thái chiến đấu bình thường)
    public static void StopBooyahEmote(Player localPlayer)
    {
        if (localPlayer == null) return;
        try
        {
            localPlayer.StopCheckBooyahEmote(true);
        }
        catch (Exception)
        {
        }
    }

    // 3. Xử lý cập nhật Toggle theo chu kỳ (tránh bị giật khựng do lặp animation mỗi frame)
    public static void UpdateSuperEmote(Player localPlayer, bool isEnabled, int currentFrame)
    {
        if (localPlayer == null) return;
        try
        {
            if (isEnabled)
            {
                // Kích hoạt lặp lại mỗi ~3 giây (180 frame ở 60 FPS) để giữ điệu nhảy liên tục
                if (currentFrame % 180 == 1)
                {
                    localPlayer.StartCheckBooyahEmote();
                    localPlayer.PlayCarni25GPDanceEmote();
                    localPlayer.RequestPlayBooyahSuperEmote(0);
                }
            }
        }
        catch (Exception)
        {
        }
    }
}

[4] BẢNG PHÂN TÍCH CƠ CHẾ HOẠT ĐỘNG & TÍNH NĂNG:
--------------------------------------------------------------------------------
1. Bình thường trong Free Fire:
   - Điệu nhảy Booyah và Super Emote tương tác chỉ được phép phát khi game kết thúc
     hoặc trong phòng chờ / sảnh chờ.
   - Trong trận đấu sinh tồn, khi đang chạy bo hoặc combat thì nút emote bị khóa hoặc
     chỉ phát được các emote cơ bản và sẽ bị ngắt ngay khi di chuyển.

2. Khi bật tính năng này:
   - Gọi trực tiếp chuỗi hàm hoạt ảnh native của nhân vật.
   - Nhảy liên tục mỗi chu kỳ ~3 giây mà không cần bấm tay vào vòng xoay emote.
   - Dùng để trêu đùa đối thủ khi knock hoặc tạo highlight cực vui nhộn.
