[1] CÁC THUỘC TÍNH VÀ CLASS CAN THIỆP TRONG FREE FIRE:
--------------------------------------------------------------------------------
1. COW.GameVarDef.MaxJumpHeight (float - Offset 0x1f0):
   - Khái niệm: Độ cao nhảy tối đa cho phép trên toàn bộ hệ thống vật lý game.
   - Mặc định trong game: ~1.0f.
   - Khi chỉnh thành: 1.2f (hoặc cao hơn tùy chỉnh)
     + Mở rộng trần giới hạn nhảy của nhân vật, cho phép vượt qua các bức tường, chướng ngại vật cao.

2. COW.GamePlay.Player.IHAAHDPHAEO (Thuộc tính PhysicalCCT - Getter Offset 0x56268c4):
   - Đối tượng điều khiển chuyển động vật lý nhân vật (Character Controller).
   - Lấy instance: localPlayer.IHAAHDPHAEO
   - Thuộc tính con: localPlayer.IHAAHDPHAEO.JumpHeight (float - Offset 0x78)
     + Lực bật nhảy trực tiếp khi người chơi bấm nút Jump.
     + Chỉnh thành: 1.2f (hoặc 1.5f, 2.0f) để nhân vật phóng lên cao vượt trội.

using System;
using COW;
using COW.GamePlay;

public static class HighJumpHelper
{
    // 1. Kích hoạt Nhảy Cao (mặc định x1.2)
    public static void EnableHighJump(Player localPlayer, float jumpScale = 1.2f)
    {
        try
        {
            COW.GameVarDef.MaxJumpHeight = jumpScale;
            if (localPlayer != null && localPlayer.IHAAHDPHAEO != null)
            {
                localPlayer.IHAAHDPHAEO.JumpHeight = jumpScale;
            }
        }
        catch (Exception)
        {
        }
    }

    // 2. Tắt Nhảy Cao (Khôi phục mặc định 1.0f)
    public static void DisableHighJump(Player localPlayer)
    {
        try
        {
            COW.GameVarDef.MaxJumpHeight = 1.0f;
            if (localPlayer != null && localPlayer.IHAAHDPHAEO != null)
            {
                localPlayer.IHAAHDPHAEO.JumpHeight = 1.0f;
            }
        }
        catch (Exception)
        {
        }
    }

    // 3. Cập nhật mỗi Frame trong Update / Draw
    public static void UpdateHighJump(Player localPlayer, bool isEnabled, float jumpScale = 1.2f)
    {
        try
        {
            if (isEnabled)
            {
                COW.GameVarDef.MaxJumpHeight = jumpScale;
                if (localPlayer != null && localPlayer.IHAAHDPHAEO != null)
                {
                    localPlayer.IHAAHDPHAEO.JumpHeight = jumpScale;
                }
            }
            else
            {
                if (COW.GameVarDef.MaxJumpHeight > 1.05f)
                {
                    COW.GameVarDef.MaxJumpHeight = 1.0f;
                    if (localPlayer != null && localPlayer.IHAAHDPHAEO != null)
                    {
                        localPlayer.IHAAHDPHAEO.JumpHeight = 1.0f;
                    }
                }
            }
        }
        catch (Exception)
        {
        }
    }
}
