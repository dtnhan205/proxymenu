[1] CÁC THUỘC TÍNH VÀ CLASS CAN THIỆP TRONG FREE FIRE:
--------------------------------------------------------------------------------
Toàn bộ các biến cấu hình camera nằm trong lớp COW.GameVarDef (Static Fields):

1. COW.GameVarDef.CameraShakeCabinFactor (float - Offset 0x3028):
   - Khái niệm: Hệ số rung lắc camera khi ngồi trong cabin/phương tiện hoặc khi chịu rung chấn nổ.
   - Mặc định trong game: ~2.0f - 5.0f.
   - Khi chỉnh thành: 1.0f (hoặc 0.0f)
     + Khử hoàn toàn rung lắc màn hình khi di chuyển, lái xe hoặc bị bom nổ gần.

2. COW.GameVarDef.CameraBackOffectLowerCutAngle (float - Offset 0x30a4):
   - Khái niệm: Góc cắt hạ thấp camera sau lưng nhân vật.
   - Khi chỉnh thành: 1.0f
     + Góc nhìn từ sau lưng nhân vật được giữ ổn định, không bị méo góc khi nhìn xuống dốc.

3. COW.GameVarDef.CameraBackOffectLowerAngleDist (float - Offset 0x30a8):
   - Khái niệm: Khoảng cách bù trừ góc nghiêng camera sau lưng.
   - Khi chỉnh thành: 1.0f
     + Giữ camera sau lưng nhân vật luôn ở khoảng cách quan sát tối ưu, không bị giật zoom lại gần.

4. COW.GameVarDef.CameraUpOffectLowerAngleDist (float - Offset 0x30ac):
   - Khái niệm: Khoảng cách góc hạ trục nâng camera.
   - Khi chỉnh thành: 1.0f
     + Giữ trục camera hướng thẳng, không bị cụp đầu xuống khi nhân vật đứng sát tường.

5. COW.GameVarDef.CameraUpOffectLowerHitDrawBackDist (float - Offset 0x30b0):
   - Khái niệm: Khoảng cách giật lùi camera khi va chạm địa hình / góc hẹp.
   - Mặc định: Camera bị giật kéo áp sát vào lưng nhân vật khi đứng gần tường hoặc trong nhà hẹp.
   - Khi chỉnh thành: 1.0f
     + Triệt tiêu hiện tượng camera bị giật zoom che khuất tầm nhìn trong nhà hoặc sau vật cản.

using System;
using COW;

public static class CameraHackHelper
{
    // 1. Kích hoạt Camera Hack (Khử rung lắc + Ổn định góc nhìn camera)
    public static void EnableCameraHack()
    {
        try
        {
            COW.GameVarDef.CameraShakeCabinFactor = 1.0f;
            COW.GameVarDef.CameraBackOffectLowerCutAngle = 1.0f;
            COW.GameVarDef.CameraBackOffectLowerAngleDist = 1.0f;
            COW.GameVarDef.CameraUpOffectLowerAngleDist = 1.0f;
            COW.GameVarDef.CameraUpOffectLowerHitDrawBackDist = 1.0f;
        }
        catch (Exception)
        {
        }
    }

    // 2. Cập nhật mỗi Frame trong Update / Draw / OnGUI
    public static void UpdateCameraHack(bool isEnabled)
    {
        try
        {
            if (isEnabled)
            {
                COW.GameVarDef.CameraShakeCabinFactor = 1.0f;
                COW.GameVarDef.CameraBackOffectLowerCutAngle = 1.0f;
                COW.GameVarDef.CameraBackOffectLowerAngleDist = 1.0f;
                COW.GameVarDef.CameraUpOffectLowerAngleDist = 1.0f;
                COW.GameVarDef.CameraUpOffectLowerHitDrawBackDist = 1.0f;
            }
        }
        catch (Exception)
        {
        }
    }
}
